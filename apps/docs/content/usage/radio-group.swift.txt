import SwiftUI

struct RadioGroupExample: View {
    @State private var value = "monthly"

    var body: some View {
        ManicatRadioGroup("Billing period", options: [
            ManicatRadioOption(value: "monthly", label: "Monthly", description: "Cancel any time."),
            ManicatRadioOption(value: "yearly", label: "Yearly", description: "Two months free."),
        ], value: $value)
    }
}

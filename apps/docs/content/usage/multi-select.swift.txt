import SwiftUI

struct MultiSelectExample: View {
    @State private var value: Set<String> = ["design", "motion"]

    var body: some View {
        ManicatMultiSelect("Skills", options: [
            ManicatMultiSelectOption(value: "design", label: "Design"),
            ManicatMultiSelectOption(value: "motion", label: "Motion"),
            ManicatMultiSelectOption(value: "data", label: "Data visualisation"),
            ManicatMultiSelectOption(value: "a11y", label: "Accessibility"),
            ManicatMultiSelectOption(value: "i18n", label: "Internationalisation"),
        ], value: $value, maxVisible: 2)
    }
}

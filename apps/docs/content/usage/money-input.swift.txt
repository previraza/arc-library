import SwiftUI

struct MoneyInputExample: View {
    @State private var amount: Double? = 1250
    @State private var currency = "USD"

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            ManicatMoneyInput("Amount", amount: $amount, currency: $currency)
            Text(amount.map { String(format: "%.2f in the currency above", $0) } ?? "Nothing charged.").font(.callout).foregroundStyle(.secondary)
        }
        .frame(maxWidth: 384)
    }
}

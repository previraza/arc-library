import SwiftUI

struct SelectExample: View {
    @State private var role: String? = "editor"

    var body: some View {
        ManicatSelect("Role", options: [
            ManicatSelectOption(value: "viewer", label: "Viewer"),
            ManicatSelectOption(value: "editor", label: "Editor"),
            ManicatSelectOption(value: "admin", label: "Admin"),
            ManicatSelectOption(value: "owner", label: "Owner", disabled: true),
        ], selection: $role, description: "What they can change.")
        .frame(maxWidth: 280)
    }
}

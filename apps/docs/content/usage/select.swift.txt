import SwiftUI

struct SelectExample: View {
    @State private var value: String? = "weekly"

    var body: some View {
        ManicatSelect("Report frequency", options: [
            ManicatSelectOption(value: "daily", label: "Daily"),
            ManicatSelectOption(value: "weekly", label: "Weekly"),
            ManicatSelectOption(value: "monthly", label: "Monthly"),
            ManicatSelectOption(value: "never", label: "Never", disabled: true),
        ], selection: $value, description: "We send it in your timezone.")
    }
}

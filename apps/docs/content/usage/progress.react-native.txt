import { useState } from "react";
import { View } from "react-native";
import { Button } from "@/components/manicat/button/button.rn";
import { Progress } from "@/components/manicat/progress/progress.rn";

export function ProgressDemo() {
  const [value, setValue] = useState(64);

  return (
    <View style={{ width: "100%", maxWidth: 384, gap: 16 }}>
      <Progress label="Migrating components" value={value} showValue />
      <View style={{ flexDirection: "row", gap: 8 }}>
        <Button size="sm" variant="secondary" onPress={() => setValue((current) => Math.max(0, current - 20))}>-20</Button>
        <Button size="sm" onPress={() => setValue((current) => Math.min(100, current + 20))}>+20</Button>
      </View>
    </View>
  );
}

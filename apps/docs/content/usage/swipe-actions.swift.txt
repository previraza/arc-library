import SwiftUI

struct SwipeActionsExample: View {
    private func message(_ name: String, _ subject: String) -> some View {
        HStack(spacing: 12) {
            Text(name).font(.subheadline.weight(.medium))
            Text(subject).font(.subheadline).foregroundStyle(.secondary)
        }
        .padding(.vertical, 8)
    }

    var body: some View {
        ManicatSwipeActions("Message actions") {
            ManicatSwipeActionsRow("Ada Lovelace",
                leading: [ManicatSwipeAction("Pin", keepRow: true, onSelect: {})],
                trailing: [ManicatSwipeAction("Archive", onSelect: {}), ManicatSwipeAction("Delete", tone: .danger, onSelect: {})]) {
                message("Ada Lovelace", "Analytical engine notes")
            }
            ManicatSwipeActionsRow("Grace Hopper", trailing: [ManicatSwipeAction("Delete", tone: .danger, onSelect: {})]) {
                message("Grace Hopper", "Compiler design")
            }
        }
    }
}

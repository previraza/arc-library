import SwiftUI

struct MentionInputExample: View {
    @State private var message = ManicatMentionValue()

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            ManicatMentionInput(value: $message, people: [
                ManicatMentionPerson(id: "1", name: "Ada", role: "Engineer"),
                ManicatMentionPerson(id: "2", name: "Lin", role: "Design"),
            ], channels: [
                ManicatMentionChannel(id: "c1", name: "general", members: 42),
                ManicatMentionChannel(id: "c2", name: "releases", members: 9),
            ])
            Text("\(message.mentions.count) mention\(message.mentions.count == 1 ? "" : "s")").font(.caption)
        }
        .frame(maxWidth: 420)
    }
}

import SwiftUI

struct MetricCardExample: View {
    @State private var revenue = 48250.0

    var body: some View {
        ManicatFlowLayout(spacing: 16, lineSpacing: 16) {
            ManicatMetricCard(label: "Monthly revenue", value: revenue, suffix: " USD", context: "vs 41,900 last month", change: "+15.2%")
            ManicatButton("Add 1,250", variant: .secondary, size: .sm) { revenue += 1250 }
        }
    }
}

import SwiftUI

struct PasswordStrengthExample: View {
    @State private var value = ""

    var body: some View {
        VStack(alignment: .leading, spacing: 16) {
            ManicatPasswordStrength("Password", password: $value)
            Text("Strength is the share of rules met, spread over four steps.").font(.callout).foregroundStyle(.secondary)
        }
        .frame(maxWidth: 384)
    }
}

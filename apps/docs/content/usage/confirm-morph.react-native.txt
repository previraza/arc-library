import { ConfirmMorph } from "@/components/manicat/confirm-morph/confirm-morph.rn";

const wait = (ms: number) => new Promise<void>((done) => setTimeout(done, ms));

export function ConfirmMorphDemo() {
  return <ConfirmMorph label="Delete" tone="danger" onConfirm={() => wait(1200)} onUndo={() => wait(600)} />;
}

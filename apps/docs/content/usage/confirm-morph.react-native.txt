import { ConfirmMorph } from "@/components/manicat/confirm-morph/confirm-morph.rn";

export function DeleteProject() {
  return (
    <ConfirmMorph
      label="Delete"
      tone="danger"
      onConfirm={() => new Promise<void>((done) => setTimeout(done, 1200))}
      onUndo={() => new Promise<void>((done) => setTimeout(done, 600))}
    />
  );
}

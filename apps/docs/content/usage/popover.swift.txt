import SwiftUI

struct PopoverExample: View {
    @State private var open = false

    var body: some View {
        ManicatPopover(isPresented: $open) {
            Text("Column settings")
        } content: {
            VStack(alignment: .leading, spacing: 4) {
                Text("Visible columns").fontWeight(.medium)
                Text("The panel stays anchored to the trigger while the trigger springs back.").foregroundStyle(.secondary)
                ManicatButton("Done", size: .sm) { open = false }.padding(.top, 12)
            }
            .frame(width: 256)
        }
    }
}

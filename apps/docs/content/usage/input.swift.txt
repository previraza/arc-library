import SwiftUI

struct InputExample: View {
    @State private var email = ""
    @State private var team = "arc-labs"
    @State private var locked = ""

    var body: some View {
        VStack(spacing: 16) {
            ManicatInput("Email", text: $email, placeholder: "ada@example.com", description: "We only use it for receipts.")
            ManicatInput("Team URL", text: $team, error: "That URL is already taken.")
            ManicatInput("Disabled", text: $locked, placeholder: "Locked").disabled(true)
        }
        .frame(maxWidth: 384)
    }
}

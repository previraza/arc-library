import SwiftUI

struct SearchFieldExample: View {
    @State private var value = ""

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            ManicatSearchField("Search", text: $value, placeholder: "Find a component")
            Text(value.isEmpty ? "Type to search." : "Searching for “\(value)”").font(.callout).foregroundStyle(.secondary)
        }
        .frame(maxWidth: 384)
    }
}

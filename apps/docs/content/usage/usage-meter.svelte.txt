<script lang="ts">
  import Button from "@/components/manicat/button/button.svelte";
  import UsageMeter from "@/components/manicat/usage-meter/usage-meter.svelte";

  let used = $state(64);
  const segments = $derived([
    { id: "db", label: "Database", value: used * 0.5 },
    { id: "media", label: "Media", value: used * 0.4 },
    { id: "logs", label: "Logs", value: used * 0.1 },
  ]);
</script>

<div style="display: grid; width: 100%; max-width: 384px; gap: 12px">
  <UsageMeter label="Workspace storage" limit={100} unit="GB" {segments} />
  <Button size="sm" variant="secondary" onclick={() => (used = used > 90 ? 20 : used + 20)}>Add 20 GB</Button>
</div>

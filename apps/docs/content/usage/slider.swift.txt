import SwiftUI

struct SliderExample: View {
    @State private var volume = 40.0
    @State private var price: ClosedRange<Double> = 20...70

    var body: some View {
        VStack(spacing: 20) {
            ManicatSlider("Volume", value: $volume)
            ManicatRangeSlider("Price", value: $price, format: { "$\(Int($0))" })
        }
        .frame(maxWidth: 340)
    }
}

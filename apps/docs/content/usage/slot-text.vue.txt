<script setup lang="ts">
import { onBeforeUnmount, onMounted, ref } from "vue";
import SlotText from "@/components/manicat/slot-text/slot-text.vue";

const amounts = [1024, 12480, 99200, 736000];
const index = ref(0);
let timer: ReturnType<typeof setInterval>;
onMounted(() => (timer = setInterval(() => (index.value = (index.value + 1) % amounts.length), 2000)));
onBeforeUnmount(() => clearInterval(timer));
</script>

<template>
  <div style="display: flex; width: 100%; flex-direction: column; align-items: center; gap: 12px">
    <p style="font-size: 28px; line-height: 37.3333px; font-weight: 600; font-variant-numeric: tabular-nums">$<SlotText :value="amounts[index]" /></p>
    <p style="font-size: 12px; line-height: 16px; color: var(--text-muted)">The last reel lands on the new digit as the length changes.</p>
  </div>
</template>

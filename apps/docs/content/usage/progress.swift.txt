import SwiftUI

struct ProgressExample: View {
    @State private var value = 64.0

    var body: some View {
        VStack(alignment: .leading, spacing: 16) {
            ManicatProgress(value: value, label: "Migrating components", showValue: true)
            HStack(spacing: 8) {
                ManicatButton("-20", variant: .secondary, size: .sm) { value = max(0, value - 20) }
                ManicatButton("+20", size: .sm) { value = min(100, value + 20) }
            }
        }
        .frame(maxWidth: 384)
    }
}

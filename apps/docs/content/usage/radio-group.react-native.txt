import { useState } from "react";
import { RadioGroup } from "@/components/manicat/radio-group/radio-group.rn";

const options = [
  { value: "monthly", label: "Monthly", description: "Cancel any time." },
  { value: "yearly", label: "Yearly", description: "Two months free." },
];

export function RadioGroupDemo() {
  const [value, setValue] = useState("monthly");

  return <RadioGroup label="Billing period" value={value} onValueChange={setValue} options={options} />;
}

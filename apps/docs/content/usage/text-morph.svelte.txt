<script lang="ts">
  import { onMount } from "svelte";
  import TextMorph from "@/components/manicat/text-morph/text-morph.svelte";

  const words = ["spacing", "grid", "rhythm", "order"];
  let index = $state(0);

  onMount(() => {
    const timer = setInterval(() => (index = (index + 1) % words.length), 2400);
    return () => clearInterval(timer);
  });
</script>

<p style="width: 100%; text-align: center; font-size: 14px; line-height: 20px; color: var(--text-muted)">
  Good interfaces are built on <TextMorph as="strong" text={words[index]} />.
</p>

import SwiftUI

struct BillingToggleExample: View {
    @State private var period = "monthly"

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            ManicatBillingToggle(value: $period)
            ManicatBillingPrice(amount: period == "monthly" ? 12 : 9.6, period: "per seat / month", was: period == "monthly" ? nil : 12, decimals: period == "monthly" ? 0 : 2)
        }
    }
}

import { useState } from "react";
import { Select } from "@/components/manicat/select/select.rn";

const options = [
  { value: "daily", label: "Daily" },
  { value: "weekly", label: "Weekly" },
  { value: "monthly", label: "Monthly" },
  { value: "never", label: "Never", disabled: true },
];

export function SelectDemo() {
  const [value, setValue] = useState("weekly");

  return <Select label="Report frequency" options={options} value={value} onValueChange={setValue} description="We send it in your timezone." />;
}

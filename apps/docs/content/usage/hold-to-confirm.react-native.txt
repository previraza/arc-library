import { useState } from "react";
import { Text, View } from "react-native";
import { HoldToConfirm } from "@/components/manicat/hold-to-confirm/hold-to-confirm.rn";

export function HoldToConfirmDemo() {
  const [confirmed, setConfirmed] = useState(false);

  return (
    <View style={{ flexDirection: "row", flexWrap: "wrap", alignItems: "center", gap: 12 }}>
      <HoldToConfirm label="Hold to delete" onConfirm={() => setConfirmed(true)} />
      {confirmed ? <Text>Deleted.</Text> : null}
    </View>
  );
}

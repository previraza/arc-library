import { useState } from "react";
import { Text, View } from "react-native";
import { HoldToConfirm } from "@/components/manicat/hold-to-confirm/hold-to-confirm.rn";

export function DeleteAccount() {
  const [deleted, setDeleted] = useState(false);
  return (
    <View style={{ gap: 8 }}>
      <HoldToConfirm label="Hold to delete" tone="danger" onConfirm={() => setDeleted(true)} />
      {deleted ? <Text>Account scheduled for deletion.</Text> : null}
    </View>
  );
}

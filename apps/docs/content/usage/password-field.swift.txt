import SwiftUI

struct PasswordFieldExample: View {
    @State private var password = ""
    @State private var short = "short"

    var body: some View {
        VStack(spacing: 16) {
            ManicatPasswordField("Password", text: $password, placeholder: "At least 12 characters")
            ManicatPasswordField("Too short", text: $short, description: "Use at least 12 characters.")
        }
        .frame(maxWidth: 384)
    }
}

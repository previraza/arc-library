import SwiftUI

struct SplitButtonExample: View {
    @State private var count = 0

    var body: some View {
        ManicatSplitButton("Publish (\(count))", actions: [
            ManicatMenuItem("Publish now") {},
            ManicatMenuItem("Schedule for later") {},
            ManicatMenuItem("Publish to a copy") {},
        ]) { count += 1 }
    }
}

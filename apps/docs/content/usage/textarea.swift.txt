import SwiftUI

struct TextareaExample: View {
    @State private var value = ""
    @State private var note = "Kept to two lines."

    var body: some View {
        VStack(spacing: 16) {
            ManicatTextarea("Changelog entry", text: $value, placeholder: "What changed?", rows: 5)
            ManicatTextarea("Short note", text: $note, description: "Two lines is enough.", rows: 2)
        }
        .frame(maxWidth: 448)
    }
}

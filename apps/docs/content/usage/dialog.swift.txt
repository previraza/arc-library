import SwiftUI

struct DialogExample: View {
    @State private var open = false
    @State private var name = "arc-library"

    var body: some View {
        ManicatButton("Open dialog") { open = true }
            .manicatDialog(isPresented: $open, title: "Rename project", description: "The slug updates everywhere it is linked.") {
                VStack(alignment: .leading, spacing: 16) {
                    ManicatInput("Project name", text: $name)
                    HStack(spacing: 8) {
                        Spacer(minLength: 0)
                        ManicatButton("Cancel", variant: .secondary) { open = false }
                        ManicatButton("Save") { open = false }
                    }
                }
            }
    }
}

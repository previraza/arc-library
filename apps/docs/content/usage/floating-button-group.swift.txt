import SwiftUI

struct FloatingButtonGroupExample: View {
    @State private var last = ""

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            ManicatFloatingButtonGroup(items: [
                .action(id: "bold", label: "Bold", shortcut: "⌘B"),
                .action(id: "italic", label: "Italic", shortcut: "⌘I"),
                .separator(id: "sep"),
                .action(id: "link", label: "Link"),
            ], label: "Editor", variant: .floating) { last = $0 }
            Text(last.isEmpty ? "" : "Pressed \(last)").font(.caption)
        }
    }
}

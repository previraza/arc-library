import SwiftUI

struct ShortcutRecorderExample: View {
    @State private var shortcut: String?

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            ManicatShortcutRecorder("Shortcut", value: $shortcut)
            Text(shortcut.map { "Recorded: \($0)" } ?? "Click, then press keys.").font(.callout).foregroundStyle(.secondary)
        }
        .frame(maxWidth: 384)
    }
}

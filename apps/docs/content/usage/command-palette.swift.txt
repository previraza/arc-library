import SwiftUI

struct CommandPaletteExample: View {
    @State private var chosen = ""

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            ManicatCommandPalette(items: [
                ManicatCommandItem(id: "new", label: "New file", group: "File", shortcut: "⌘N"),
                ManicatCommandItem(id: "open", label: "Open…", group: "File", shortcut: "⌘O"),
                ManicatCommandItem(id: "theme", label: "Toggle theme", description: "Switch light and dark", group: "View"),
            ]) { chosen = $0.label }
            Text(chosen.isEmpty ? "" : "Chose \(chosen)").font(.caption)
        }
        .frame(maxWidth: 420)
    }
}

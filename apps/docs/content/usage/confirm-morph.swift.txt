import SwiftUI

struct ConfirmMorphExample: View {
    @State private var state: ManicatConfirmState = .idle

    var body: some View {
        ManicatConfirmMorph("Delete", state: $state, tone: .danger, onUndo: { try await Task.sleep(nanoseconds: 600_000_000) }) {
            try await Task.sleep(nanoseconds: 1_200_000_000)
        }
    }
}

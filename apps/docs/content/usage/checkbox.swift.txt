import SwiftUI

struct CheckboxExample: View {
    @State private var releases: ManicatCheckedState = .indeterminate
    @State private var digest = false

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            ManicatCheckbox(state: $releases, label: "Email me about releases")
            ManicatCheckbox(isOn: $digest, label: "Include a digest", description: "One email a week, never more.")
            ManicatCheckbox(isOn: .constant(false), label: "Locked").disabled(true)
        }
    }
}

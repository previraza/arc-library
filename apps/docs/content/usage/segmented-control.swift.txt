import SwiftUI

struct SegmentedControlExample: View {
    @State private var value = "grid"

    var body: some View {
        ManicatSegmentedControl(
            options: [ManicatSegment(value: "grid", label: "Grid"), ManicatSegment(value: "list", label: "List"), ManicatSegment(value: "map", label: "Map")],
            value: $value,
            label: "Layout"
        )
    }
}

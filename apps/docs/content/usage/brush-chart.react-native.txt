import { BrushChart } from "@/components/manicat/brush-chart/brush-chart.rn";

const DAY = 86_400_000;
const start = Date.UTC(2026, 6, 1, 12);
const data = Array.from({ length: 60 }, (_, index) => ({ date: start + index * DAY, value: 1200 + Math.round(260 * Math.sin(index / 5.5) + index * 7) }));

export function Users() {
  return <BrushChart data={data} label="Daily active users" unit="users" annotations={[{ date: data[20].date, label: "v2 launch" }]} />;
}

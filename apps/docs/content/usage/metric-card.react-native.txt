import { useState } from "react";
import { View } from "react-native";
import { Button } from "@/components/manicat/button/button.rn";
import { MetricCard } from "@/components/manicat/metric-card/metric-card.rn";

export function MetricCardDemo() {
  const [revenue, setRevenue] = useState(48250);

  return (
    <View style={{ flexDirection: "row", flexWrap: "wrap", alignItems: "center", gap: 16 }}>
      <MetricCard label="Monthly revenue" value={revenue} suffix=" USD" context="vs 41,900 last month" change="+15.2%" />
      <Button size="sm" variant="secondary" onPress={() => setRevenue((value) => value + 1250)}>Add 1,250</Button>
    </View>
  );
}

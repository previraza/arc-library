import SwiftUI

struct SwitchExample: View {
    @State private var reduced = true
    @State private var always = false

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            ManicatSwitch(isOn: $reduced, label: "Reduced motion")
            ManicatSwitch(isOn: $always, label: "Always on")
        }
    }
}

import SwiftUI

struct DropdownMenuExample: View {
    @State private var opened: String?

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            ManicatDropdownMenu("Row actions", items: [
                ManicatMenuItem("Rename") { opened = "Rename" },
                ManicatMenuItem("Duplicate") {},
                ManicatMenuItem("Move to archive", separatorBefore: true) {},
                ManicatMenuItem("Delete", destructive: true) { opened = "Delete" },
            ])
            if let opened { Text("Selected: \(opened)").font(.callout).foregroundStyle(.secondary) }
        }
    }
}
